%
O1630 (MACRO CONE EXTERNO COM FRESA 90 GRAUS EM PASSADAS - CHAMADA POR G65 P1630) ;
(FRESA 90 GRAUS = FRESA DE TOPO RETA - O CORRETOR DE COMPRIMENTO FICA NA PONTA DA FRESA) ;
(CONE NA PECA DE FORA - DIAMETRO MENOR NO TOPO E MAIOR NA BASE - O BRUTO TEM O DIAMETRO D DA BASE) ;
(A QUINA DE BAIXO DA FRESA ANDA SOBRE A LINHA DO CONE - DE CIMA PARA BAIXO - UMA VOLTA COMPLETA EM CADA POSICAO - SENTIDO HORARIO G2) ;
(SEMPRE FAZ VARIAS PASSADAS - O Q DEFINE QUANTO A FRESA DESCE EM CADA VOLTA - Q PEQUENO = MELHOR ACABAMENTO) ;
(O CONE E DEFINIDO PELO ANGULO B OU PELO DIAMETRO DO TOPO A) ;
(SE INFORMAR B O DIAMETRO A E CALCULADO E NAO PRECISA INFORMAR) ;
(LETRAS DO G65 P1630) ;
(X = X DO CENTRO DO CONE) ;
(Y = Y DO CENTRO DO CONE) ;
(D = DIAMETRO MAIOR DO CONE - NA BASE - IGUAL AO DIAMETRO DO BRUTO) ;
(B = ANGULO DO CONE EM RELACAO A FACE DE CIMA - DE 1 ATE 89) ;
(A = DIAMETRO MENOR DO CONE NO TOPO - NAO USA SE INFORMAR B) ;
(W = Z DO TOPO DO CONE) ;
(Z = Z DA BASE DO CONE) ;
(Q = PASSE EM Z - QUANTO A FRESA DESCE EM CADA VOLTA) ;
(T = DIAMETRO DA FRESA) ;
(R = PLANO DE SEGURANCA ACIMA DO TOPO) ;
(F = AVANCO DO CORTE) ;
(EXEMPLO COM ANGULO - G65 P1630 X0 Y0 D60. B60 W0 Z-20. Q0.3 T25. R5. F800.) ;
(EXEMPLO COM DIAMETRO DO TOPO - G65 P1630 X0 Y0 D60. A44. W0 Z-8. Q0.5 T28. R5. F800.) ;
(VARIAVEIS RECEBIDAS DO G65) ;
(#24 = X  X DO CENTRO DO CONE) ;
(#25 = Y  Y DO CENTRO DO CONE) ;
(#7 = D  DIAMETRO MAIOR DO CONE) ;
(#2 = B  ANGULO DO CONE EM RELACAO A FACE DE CIMA) ;
(#1 = A  DIAMETRO MENOR DO CONE) ;
(#23 = W  Z DO TOPO DO CONE) ;
(#26 = Z  Z DA BASE DO CONE) ;
(#17 = Q  PASSE EM Z A CADA VOLTA) ;
(#20 = T  DIAMETRO DA FRESA) ;
(#18 = R  PLANO DE SEGURANCA ACIMA DO TOPO) ;
(#9 = F  AVANCO DO CORTE) ;
(VARIAVEIS DE CALCULO DA MACRO) ;
(#100 = RAIO DO BRUTO = RAIO MAIOR DO CONE) ;
(#101 = RAIO MENOR DO CONE NO TOPO) ;
(#103 = Z DA FRESA NA VOLTA ATUAL) ;
(#104 = ABERTURA DO CONE = RAIO QUE AUMENTA PARA CADA 1 DE DESCIDA) ;
(#107 = RAIO DO CENTRO DA FRESA NA VOLTA ATUAL) ;
(#108 = 0 ANTES DA PRIMEIRA VOLTA - 1 DEPOIS) ;
IF [#20 LE 0] THEN #3000 = 281 (T INVALIDO) ;
IF [#26 GE #23] THEN #3000 = 282 (Z FINAL TEM QUE SER MENOR QUE W) ;
IF [#7 LE 0] THEN #3000 = 283 (D INVALIDO) ;
IF [#17 LE 0] THEN #3000 = 287 (Q INVALIDO - INFORME O PASSE EM Z) ;
IF [#2 EQ #0] GOTO 4 ;
(CONE DEFINIDO PELO ANGULO B) ;
IF [#2 GE 90] THEN #3000 = 289 (B INVALIDO) ;
IF [#2 LT 1] THEN #3000 = 289 (B INVALIDO) ;
#104 = 1 / TAN[#2] ;
#1 = #7 - 2 * [#23 - #26] * #104 ;
IF [#1 LT 0] THEN #3000 = 284 (B MUITO ABERTO - O TOPO PASSOU DO CENTRO - DIMINUA A ALTURA OU AUMENTE B) ;
GOTO 5 ;
(CONE DEFINIDO PELO DIAMETRO A) ;
N4 IF [#1 LT 0] THEN #3000 = 284 (A INVALIDO) ;
IF [#1 GE #7] THEN #3000 = 284 (A TEM QUE SER MENOR QUE D) ;
#104 = [#7 - #1] / 2 / [#23 - #26] ;
N5 IF [#104 * #17 GT #20] THEN #3000 = 290 (Q MUITO GRANDE PARA ESSE ANGULO - AVANCO RADIAL MAIOR QUE A FRESA) ;
#100 = #7 / 2 ;
#101 = #1 / 2 ;
#103 = #23 + #17 ;
#108 = 0 ;
(N10 = CADA POSICAO DA FRESA - COMECA NO TOPO E DESCE PELA LINHA DO CONE) ;
N10 #103 = #103 - #17 ;
IF [#103 LT #26] THEN #103 = #26 ;
#107 = #101 + [#23 - #103] * #104 + #20 / 2 ;
IF [#108 NE 0] GOTO 20 ;
(PRIMEIRA VOLTA - ENTRA POR FORA DO BRUTO E ENCOSTA NO TOPO DO CONE) ;
G0 X[#24 + #100 + #20 / 2 + 2] Y#25 ;
G0 Z[#23 + #18] ;
G1 Z#103 F#9 ;
G1 X[#24 + #107] F#9 ;
#108 = 1 ;
GOTO 30 ;
N20 G1 X[#24 + #107] Z#103 F#9 ;
N30 G2 X[#24 + #107] Y#25 I[-#107] J0 F#9 ;
IF [#103 GT #26] GOTO 10 ;
G1 X[#24 + #100 + #20 / 2 + 2] F#9 ;
G0 Z[#23 + #18] ;
M99 ;
%
